% =====================================================================
%  SHARED PAGE BACKDROP — background, left bar, vertical wordmark,
%  full-page guilloché engraving and the top-right logo mark.
%  Call inside a tikzpicture[remember picture,overlay]. #1 = bar wordmark.
% =====================================================================
\newcommand{\pagebackdrop}[1]{%
  \fill[bg] (current page.south west) rectangle (current page.north east);
  \fill[band] (current page.south west) rectangle ($(current page.north west)+(\barwidth,0)$);
  \node[rotate=90, text=barword, anchor=center] at
       ($(current page.west)+({\barwidth/2},0)$)
       {\resizebox{\paperheight}{!}{\bfseries #1}};
  \begin{scope}
    \clip (current page.south west) rectangle (current page.north east);
    \begin{scope}[shift={(current page.center)}]
      \foreach \col/\off in {guilB/0, guilA/3}{
        \foreach \a in {0,6,...,359}{
          \draw[\col,line width=0.1pt,opacity=0.14]
            (\a+\off:1.4) .. controls (\a+\off+9:6.5) and (\a+\off+16:12.5) .. (\a+\off+24:18.5);
        }
      }
      \foreach \ph/\col in {0/guilC, 6/guilB}{
        \draw[\col,line width=0.12pt,opacity=0.19,rotate=\ph]
          plot[domain=0:360,samples=560,smooth cycle,variable=\a]
          ({(10.6+1.5*cos(30*\a))*cos(\a)},{(10.6+1.5*cos(30*\a))*sin(\a)});
      }
      \foreach \ph/\col in {0/guilB, 7.5/guilA}{
        \draw[\col,line width=0.12pt,opacity=0.22,rotate=\ph]
          plot[domain=0:360,samples=520,smooth cycle,variable=\a]
          ({(7.8+1.3*cos(24*\a))*cos(\a)},{(7.8+1.3*cos(24*\a))*sin(\a)});
      }
      \foreach \ph/\col in {0/guilA, 10/guilB}{
        \draw[\col,line width=0.12pt,opacity=0.23,rotate=\ph]
          plot[domain=0:360,samples=460,smooth cycle,variable=\a]
          ({(5.2+1.0*cos(18*\a))*cos(\a)},{(5.2+1.0*cos(18*\a))*sin(\a)});
      }
      \foreach \col/\rot in {guilB/0, guilA/6}{
        \draw[\col,line width=0.12pt,opacity=0.26,rotate=\rot]
          plot[domain=0:1800,samples=720,smooth,variable=\a]
          ({4.0*cos(\a)+2.2*cos(5*\a)},{4.0*sin(\a)-2.2*sin(5*\a)});
      }
    \end{scope}
  \end{scope}
  \node[anchor=north east, text=logogrey] at ($(current page.north east)+(-1.1,-1.15)$)
        {\fontsize{58}{58}\selectfont\bfseries \logoMark};
  \draw[accent,line width=1.3pt]
        ($(current page.north east)+(-3.9,-2.15)$) -- ($(current page.north east)+(-1.1,-2.15)$);%
}
