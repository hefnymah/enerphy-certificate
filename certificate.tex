\newcommand{\drawbird}[4]{%
  \begin{scope}[shift={#1},scale=#2,rotate=#3]
    \fill[#4]
      % --- beak & head (pointing left) ---
      (-1.58,0.05)
        .. controls (-1.38,0.10) and (-1.22,0.10) .. (-1.06,0.08)
      % --- neck rising to the shoulder ---
        .. controls (-0.80,0.10) and (-0.58,0.20) .. (-0.40,0.36)
      % --- leading edge of the raised wing, up to the tip ---
        .. controls (-0.28,0.58) and (-0.10,0.88) .. (0.12,1.05)
      % --- wing tip, trailing edge coming back down ---
        .. controls (0.18,0.86) and (0.17,0.60) .. (0.15,0.40)
      % --- back edge sweeping toward the tail ---
        .. controls (0.22,0.30) and (0.38,0.22) .. (0.58,0.16)
      % --- tail & trailing legs (to the right) ---
        .. controls (0.90,0.11) and (1.20,0.05) .. (1.52,-0.03)
        .. controls (1.18,-0.03) and (0.82,-0.01) .. (0.52,0.01)
      % --- belly / underside back to the neck ---
        .. controls (0.24,0.01) and (-0.02,-0.01) .. (-0.28,-0.03)
        .. controls (-0.58,-0.05) and (-0.88,-0.01) .. (-1.08,0.01)
        .. controls (-1.24,0.01) and (-1.42,0.02) .. (-1.58,0.05)
      -- cycle;
  \end{scope}}

  \begin{scope}
    \clip (current page.south west) rectangle (current page.north east);
    \begin{scope}[shift={(current page.center)}]

      \foreach \col/\off in {guilB/0, guilA/3}{
        \foreach \a in {0,6,...,359}{
          \draw[\col,line width=0.1pt,opacity=0.14]
            (\a+\off:1.4)
              .. controls (\a+\off+9:6.5) and (\a+\off+16:12.5)
              .. (\a+\off+24:18.5);
        }
      }

      % --- big outer petal ring ---------------------------------------
      \foreach \ph/\col in {0/guilC, 6/guilB}{
        \draw[\col,line width=0.12pt,opacity=0.19,rotate=\ph]
          plot[domain=0:360,samples=560,smooth cycle,variable=\a]
          ({(10.6+1.5*cos(30*\a))*cos(\a)},{(10.6+1.5*cos(30*\a))*sin(\a)});
      }
      % --- outer petal ring (interwoven colours) ----------------------

        \draw[\col,line width=0.12pt,opacity=0.22,rotate=\ph]
          plot[domain=0:360,samples=520,smooth cycle,variable=\a]

      % --- middle ring ------------------------------------------------

          plot[domain=0:360,samples=460,smooth cycle,variable=\a]
          ({(5.2+1.0*cos(18*\a))*cos(\a)},{(5.2+1.0*cos(18*\a))*sin(\a)});
      }
      % --- inner hypotrochoid star rosette ----------------------------
